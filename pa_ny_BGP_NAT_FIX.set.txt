# PA_NY - apply in configure mode, then commit
# Goal: use OSPF only toward NY inside routers, BGP only toward provider, and add PAT.

configure

# Correct identity
set deviceconfig system hostname FW_NY

# Interface roles assumed from your existing config:
# ethernet1/1 = inside to R1_NY 10.101.6.6/30
# ethernet1/2 = inside to R2_NY 10.101.6.10/30
# ethernet1/3 = WAN to R_WAN 10.101.6.13/30

# Clean bad static route from old XML if present
delete network virtual-router VR_MAIN routing-table ip static-route NY_LAN2

# Keep only the default route to provider; NY LAN routes should be learned through OSPF from R1/R2.
set network virtual-router VR_MAIN routing-table ip static-route Default-to-provider destination 0.0.0.0/0 interface ethernet1/3 nexthop ip-address 10.101.6.14 metric 10

# OSPF: enable only toward internal routers, not provider
set network virtual-router VR_MAIN protocol ospf enable yes
set network virtual-router VR_MAIN protocol ospf router-id 10.101.7.3
set network virtual-router VR_MAIN protocol ospf area 0.0.0.0 type normal
set network virtual-router VR_MAIN protocol ospf area 0.0.0.0 interface ethernet1/1 enable yes passive no link-type p2p
set network virtual-router VR_MAIN protocol ospf area 0.0.0.0 interface ethernet1/2 enable yes passive no link-type p2p
delete network virtual-router VR_MAIN protocol ospf area 0.0.0.0 interface ethernet1/3

# BGP to provider
set network virtual-router VR_MAIN protocol bgp enable yes
set network virtual-router VR_MAIN protocol bgp router-id 10.101.7.3
set network virtual-router VR_MAIN protocol bgp local-as 65100
set network virtual-router VR_MAIN protocol bgp install-route yes
set network virtual-router VR_MAIN protocol bgp auth-profile BGP_NY secret Cisco
set network virtual-router VR_MAIN protocol bgp peer-group Provider type ebgp
set network virtual-router VR_MAIN protocol bgp peer-group Provider enable yes
set network virtual-router VR_MAIN protocol bgp peer-group Provider peer ISP enable yes
set network virtual-router VR_MAIN protocol bgp peer-group Provider peer ISP peer-as 65200
set network virtual-router VR_MAIN protocol bgp peer-group Provider peer ISP peer-address ip 10.101.6.14
set network virtual-router VR_MAIN protocol bgp peer-group Provider peer ISP local-address interface ethernet1/3 ip 10.101.6.13/30
set network virtual-router VR_MAIN protocol bgp peer-group Provider peer ISP connection-options authentication BGP_NY

# Redistribute internal OSPF routes into BGP so BRU/provider can learn NY LANs
set network virtual-router VR_MAIN protocol redist-profile REDIST_NY_LANS filter type ospf
set network virtual-router VR_MAIN protocol redist-profile REDIST_NY_LANS priority 10 action redist
set network virtual-router VR_MAIN protocol bgp redist-rules REDIST_NY_LANS enable yes address-family-identifier ipv4 set-origin igp

# NAT/PAT for NY + Chicago behind FW_NY WAN interface
set rulebase nat rules PAT_NY_CHI from INSIDE to WAN source any destination any service any source-translation dynamic-ip-and-port interface-address interface ethernet1/3 ip 10.101.6.13

# Security policy for inside users to provider/WAN
set rulebase security rules ALLOW_INSIDE_TO_WAN from INSIDE to WAN source any destination any application any service application-default action allow

commit
